#include <Geode/Geode.hpp>
#include <Geode/modify/PlayLayer.hpp>
#include <Geode/modify/EndLevelLayer.hpp>

using namespace geode::prelude;

class $modify(MyPlayLayer, PlayLayer) {
    struct Fields {
        std::array<bool, 3> m_coinArr = {};
    };

    bool isRobtop(GJGameLevel* level) {
        int levelID = level->m_levelID.value();
        if (levelID < 50) return true;

        if (levelID >= 1001 && levelID <= 1003) return true;
        if (levelID >= 4001 && levelID <= 4003) return true;
        return levelID >= 5001 && levelID <= 5004 || levelID == 3001;
    }

    bool hasCoin(int n) {
        auto coinKey = this->m_level->getCoinKey(n);
        auto statsMng = GameStatsManager::sharedState();
        if (!coinKey || !statsMng) return false;

        if (isRobtop(m_level)) return statsMng->hasSecretCoin(coinKey);
        if (m_level->areCoinsVerified()) return statsMng->hasUserCoin(coinKey);

        return statsMng->hasPendingUserCoin(coinKey);
    }
};

class $modify(EndLevelLayer) {
	void customSetup() {
		EndLevelLayer::customSetup();

        auto playLayer = static_cast<MyPlayLayer*>(PlayLayer::get());
        auto isRobtop = playLayer->isRobtop(playLayer->m_level);

		auto mainLayer = this->getChildByID("main-layer");
        const char* frameName = isRobtop ? "SecretCoin1.png"_spr : "UserCoin1.png"_spr;

		for (int i = 1; i <= 3; i++) {
			auto coinNode = fmt::format("coin-{}-sprite", i);
			auto blueNode = fmt::format("blue-{}-sprite", i);
			auto oldSpr = mainLayer->getChildByID(coinNode);

			if (oldSpr && playLayer->m_fields->m_coinArr[i-1]) {
                typeinfo_cast<CCSprite*>(oldSpr)->setColor({ 255, 255, 255 });
                auto newSpr = CCSprite::createWithSpriteFrameName(frameName);
                newSpr->setPosition(oldSpr->getPosition());
                mainLayer->addChild(newSpr, 10);
                newSpr->setID(blueNode);
			}
		}
	}
};

class $modify(CCSprite) {
    void setDisplayFrame(CCSpriteFrame* newFrame) {
        if (this->getID() == "blue-coin"_spr) {
            auto playLayer = static_cast<MyPlayLayer*>(PlayLayer::get());
            bool isRobtop = playLayer->isRobtop(playLayer->m_level);
            std::string coinType = playLayer ? "Secret" : "User";

            if (!newFrame) {
                auto coinName = playLayer ? "SecretCoin1.png"_spr : "UserCoin1.png"_spr;
                auto coinSpr = CCSprite::createWithSpriteFrameName(coinName);
                if (coinSpr) CCSprite::setDisplayFrame(coinSpr->displayFrame());
                return;
            }

            for (int i = 1; i <= 4; i++) {
                std::string coinName = fmt::format("{}Coin{}.png"_spr, coinType, i);
                auto coinSpr = CCSprite::createWithSpriteFrameName(coinName.c_str());
                auto coinFrame = coinSpr->displayFrame();

                if (coinFrame && coinFrame->getTexture() == newFrame->getTexture()) {
                    CCSprite::setDisplayFrame(newFrame);
                    return;
                }
            }

            return;
        }

        CCSprite::setDisplayFrame(newFrame);
    }
};

class $modify(GameObject) {
    void playDestroyObjectAnim(GJBaseGameLayer* gameLayer) {
        if (m_objectID == 142 || m_objectID == 1329) {
            int coinIndex = -1;
            std::vector<GameObject*> levelCoins;
            auto playLayer = static_cast<MyPlayLayer*>(PlayLayer::get());

            for (auto gameObj : CCArrayExt<GameObject*>(playLayer->m_objects)) {
                if (gameObj->m_objectID == 142 || gameObj->m_objectID == 1329) {
                    levelCoins.push_back(gameObj);
                }
            }

            std::sort(levelCoins.begin(), levelCoins.end(), 
                [](GameObject* a, GameObject* b) {
                    float ax = a->getPositionX(), ay = a->getPositionY();
                    float bx = b->getPositionX(), by = b->getPositionY();
                    return (ax * ax + ay * ay) < (bx * bx + by * by);
                }
            );

            for (size_t i = 0; i < levelCoins.size(); i++) {
                if (levelCoins[i] == this) {
                    playLayer->m_fields->m_coinArr[i] = true;
                    coinIndex = static_cast<int>(i + 1);
                }
            }

            if (coinIndex == -1) coinIndex = 1;
            CCNode* parentNode = this->getParent();
            std::vector<CCNode*> prevChildren;

            for (auto childNode : parentNode->getChildrenExt()) {
                prevChildren.push_back(childNode);
            }

            GameObject::playDestroyObjectAnim(gameLayer);
            if (!playLayer->hasCoin(coinIndex)) return;

            for (auto childNode : parentNode->getChildrenExt()) {
                bool hasExisted = false;
                for (auto& n : prevChildren) {
                    if (n == childNode) {
                        hasExisted = true;
                        break;
                    }
                }

                if (hasExisted) continue;
                // std::vector<CCNode*> objStack{childNode};

                // while (!objStack.empty()) {
                //     auto currentNode = objStack.back();

                //     objStack.pop_back();
                //     if (!currentNode) continue;

                    if (auto coinSpr = typeinfo_cast<CCSprite*>(childNode)) {
                        if (typeinfo_cast<GameObject*>(childNode) != this) continue;

                        CCAnimation* coinAnim = CCAnimation::create();
                        CCSpriteFrame* firstFrame = nullptr;
                        std::string coinType = this->m_objectID == 142 ? "Secret" : "User";

                        for (int i = 1; i <= 4; i++) {
                            auto coinName = fmt::format("{}Coin{}.png"_spr, coinType, i);
                            auto tempSpr = CCSprite::createWithSpriteFrameName(coinName.c_str());

                            if (auto coinFrame = tempSpr->displayFrame()) {
                                if (!firstFrame) firstFrame = coinFrame;
                                coinAnim->addSpriteFrame(coinFrame);
                            }
                        }

                        if (coinAnim->getFrames() && coinAnim->getFrames()->count() > 0) {
                            coinAnim->setDelayPerUnit(0.05f);
                            if (firstFrame) coinSpr->setDisplayFrame(firstFrame);
                            coinSpr->runAction(CCRepeatForever::create(CCAnimate::create(coinAnim)));

                            coinSpr->setColor({ 255, 255, 255 });
                            coinSpr->setID("blue-coin"_spr);
                        }
                    }

                //     for (auto stackChild : currentNode->getChildrenExt()) objStack.push_back(stackChild);
                // }
            }

            return;
        }

        GameObject::playDestroyObjectAnim(gameLayer);
    }
};

class $modify(EffectGameObject) {
    void customSetup() {
        EffectGameObject::customSetup();

        if (this->getChildByID("blue-coin"_spr)) return;
        if (m_objectID != 142 && m_objectID != 1329) return;

        auto playLayer = static_cast<MyPlayLayer*>(PlayLayer::get());
        if (!playLayer || !playLayer->m_level) return;

        this->m_addToNodeContainer = true;
        static CCSpriteFrame* blueFirst = nullptr;
        static CCAnimation* blueAnim = CCAnimation::create();
        std::string coinType = m_objectID == 142 ? "Secret" : "User";

        for (int i = 1; i <= 4; i++) {
            std::string coinName = fmt::format("{}Coin{}.png"_spr, coinType, i);
            auto coinSpr = CCSprite::createWithSpriteFrameName(coinName.c_str());

            auto coinFrame = coinSpr->displayFrame();
            if (!blueFirst) blueFirst = coinFrame;
            blueAnim->addSpriteFrame(coinFrame);
        }
            
        blueAnim->setDelayPerUnit(0.1f);
        blueFirst->retain();
        blueAnim->retain();

        auto applyAnim = [](CCSprite* coinSpr) {
            if (!coinSpr) return;
            coinSpr->stopAllActions();

            if (blueFirst) coinSpr->setDisplayFrame(blueFirst);
            coinSpr->runAction(CCRepeatForever::create(CCAnimate::create(blueAnim)));
            
            coinSpr->setColor({ 255, 255, 255 });
            coinSpr->setID("blue-coin"_spr);
        };

        std::vector<EffectGameObject*> levelCoins;
        CCArrayExt<GameObject*> objArr = playLayer->m_objects;

        for (auto* gameObj : objArr) {
            if (gameObj->m_objectID != 142 && gameObj->m_objectID != 1329) continue;
            levelCoins.push_back(typeinfo_cast<EffectGameObject*>(gameObj));
        }

        std::sort(levelCoins.begin(), levelCoins.end(), 
            [](EffectGameObject* a, EffectGameObject* b) {
                float ax = a->getPositionX(), ay = a->getPositionY();
                float bx = b->getPositionX(), by = b->getPositionY();
                return (ax * ax + ay * ay) < (bx * bx + by * by);
            }
        );

        for (int i = 0; i < levelCoins.size(); i++) {
            if (playLayer->hasCoin(i + 1)) {
                auto coinSpr = typeinfo_cast<CCSprite*>(levelCoins[i]);
                applyAnim(coinSpr);
            }
        }

        blueFirst->release();
        blueAnim->release();
    }
};
